(function(){
  const titleInput = document.getElementById('cnTitle');
  const editor = document.getElementById('cnEditor');
  const card = document.querySelector('#cnPreviewFrame .cn-card');
  const cardTitle = document.querySelector('#cnPreviewFrame .cn-card-title');
  const cardBody = document.querySelector('#cnPreviewFrame .cn-card-body');
  const frame = document.getElementById('cnPreviewFrame');
  function syncPreview(){
    if (cardTitle) cardTitle.textContent = titleInput.value || '未片';
    if (cardBody) cardBody.innerHTML = editor.innerHTML;
  }
  if (titleInput) titleInput.addEventListener('input', syncPreview);
  if (editor) editor.addEventListener('input', syncPreview);
  document.querySelectorAll('.cn-toolbar .tb-btn[data-cmd]').forEach(btn => {
    btn.addEventListener('click', () => {
      const cmd = btn.dataset.cmd;
      editor.focus();
      if (cmd === 'h1' || cmd === 'h2' || cmd === 'h3') document.execCommand('formatBlock', false, cmd.toUpperCase());
      else if (cmd === 'bold') document.execCommand('bold');
      else if (cmd === 'italic') document.execCommand('italic');
      else if (cmd === 'ul') document.execCommand('insertUnorderedList');
      else if (cmd === 'ol') document.execCommand('insertOrderedList');
      else if (cmd === 'marker') document.execCommand('hiliteColor', false, '#FFE066');
      syncPreview();
      document.querySelectorAll('.cn-toolbar .tb-btn[data-cmd="h1"],.cn-toolbar .tb-btn[data-cmd="h2"],.cn-toolbar .tb-btn[data-cmd="h3"]').forEach(b => b.classList.remove('active'));
      if (cmd === 'h1' || cmd === 'h2' || cmd === 'h3') btn.classList.add('active');
    });
  });
  const smartBtn = document.querySelector('.cn-toolbar .tb-btn[title="AI 虐"]');
  if (smartBtn) smartBtn.addEventListener('click', () => {
    const txt = (editor.innerText || '').trim();
    if (!txt) return;
    const lines = txt.split(/\n+/).filter(Boolean);
    editor.innerHTML = lines.map(l => {
      if (l.length > 28) return '<h2>' + l + '</h2>';
      if (/^[\u4e00-\u9fa5]/.test(l) && l.length <= 14) return '<h1>' + l + '</h1>';
      return '<p>' + l + '</p>';
    }).join('');
    syncPreview();
  });
  const checkBtn = document.querySelector('.cn-toolbar .tb-btn[title="违"]');
  if (checkBtn) checkBtn.addEventListener('click', () => { showToast('扫, 未违'); });
  document.querySelectorAll('#cnTplGrid .cn-tpl-card').forEach(t => {
    t.addEventListener('click', () => {
      document.querySelectorAll('#cnTplGrid .cn-tpl-card').forEach(x => x.classList.remove('active'));
      t.classList.add('active');
      const tpl = t.dataset.tpl;
      card.className = 'cn-card cn-tpl-' + tpl;
      if (tpl === 'dark') card.style.color = '#fff'; else card.style.color = '';
    });
  });
  const r34 = document.getElementById('cnRatio34');
  const r35 = document.getElementById('cnRatio35');
  if (r34 && r35) {
    r34.addEventListener('click', () => { r34.classList.add('active'); r35.classList.remove('active'); frame.style.aspectRatio = '3/4'; });
    r35.addEventListener('click', () => { r35.classList.add('active'); r34.classList.remove('active'); frame.style.aspectRatio = '3/5'; });
  }
  document.querySelectorAll('.cn-size-row').forEach(row => {
    const btns = row.querySelectorAll('.cn-size-btn');
    btns.forEach((b, i) => b.addEventListener('click', () => {
      btns.forEach(x => x.classList.remove('active'));
      b.classList.add('active');
      const sizes = [12, 13.5, 14.5, 15.5, 16.5];
      if (row.parentElement.querySelector('.sec-title').textContent.includes('趾')) {
        cardBody.style.fontSize = sizes[i] + 'px';
      }
    }));
  });
  function wrapLines(ctx, text, maxW){
    if (!text) return [];
    const out = []; let line = '';
    for (const ch of (text || '').split('')) {
      const t = line + ch;
      if (ctx.measureText(t).width > maxW && line) { out.push(line); line = ch; }
      else line = t;
    }
    if (line) out.push(line);
    return out;
  }
  function wrapText2(ctx, text, maxW, x, y, lineH){
    const lines = wrapLines(ctx, text, maxW);
    lines.forEach(line => { ctx.fillText(line, x, y); y += lineH; });
    return y;
  }
  const dl = document.getElementById('cnDownloadBtn');
  if (dl) dl.addEventListener('click', () => {
    const w = 1080, h = frame.style.aspectRatio === '3/4' ? 1440 : 1800;
    const c = document.createElement('canvas'); c.width = w; c.height = h;
    const ctx = c.getContext('2d');
    const tplClass = Array.from(card.classList).find(c => c.startsWith('cn-tpl-')) || 'cn-tpl-default';
    const bgMap = { 'cn-tpl-default':'#ffffff','cn-tpl-mint':'#E8F5E9','cn-tpl-peach':'#FFF1E6','cn-tpl-cream':'#FFF8E1','cn-tpl-dark':'#1a1a1a','cn-tpl-lavender':'#EDE7F6' };
    ctx.fillStyle = bgMap[tplClass] || '#fff';
    ctx.fillRect(0, 0, w, h);
    ctx.fillStyle = tplClass === 'cn-tpl-dark' ? '#fff' : '#1a1a1a';
    ctx.textBaseline = 'top';
    const pad = 80;
    let y = 110;
    const badge = card.querySelector('.cn-card-badge');
    if (badge) {
      ctx.font = '600 28px -apple-system, "PingFang SC", "Microsoft YaHei", sans-serif';
      const tw = ctx.measureText(badge.textContent).width + 36;
      ctx.fillStyle = tplClass === 'cn-tpl-dark' ? 'rgba(255,255,255,.15)' : 'rgba(0,0,0,.08)';
      ctx.fillRect(pad, y, tw, 44);
      ctx.fillStyle = tplClass === 'cn-tpl-dark' ? '#fff' : '#1a1a1a';
      ctx.fillText(badge.textContent, pad + 18, y + 10);
      y += 80;
    }
    const title = titleInput.value || '未片';
    ctx.font = '800 56px -apple-system, "PingFang SC", "Microsoft YaHei", sans-serif';
    y = wrapText2(ctx, title, w - pad * 2, pad, y, 70);
    y += 24;
    const bodyEl = cardBody.cloneNode(true);
    bodyEl.querySelectorAll('h1, h2, h3').forEach(h => {
      const sz = h.tagName === 'H1' ? 50 : h.tagName === 'H2' ? 44 : 38;
      ctx.font = '700 ' + sz + 'px -apple-system, "PingFang SC", "Microsoft YaHei", sans-serif';
      y = wrapText2(ctx, h.textContent, w - pad * 2, pad, y, sz * 1.5);
      y += 14;
    });
    ctx.font = '400 32px -apple-system, "PingFang SC", "Microsoft YaHei", sans-serif';
    bodyEl.querySelectorAll('p, li').forEach(p => {
      wrapLines(ctx, p.textContent, w - pad * 2).forEach(line => {
        if (y > h - 100) return;
        ctx.fillText(line, pad, y);
        y += 50;
      });
      y += 8;
    });
    ctx.font = '400 22px -apple-system, sans-serif';

// ====== Tab 8: 品图 ======
(function(){
  const cv = document.getElementById('biPreviewCanvas');
  if (!cv) return;
  const ctx = cv.getContext('2d');
  const W = cv.width, H = cv.height;
  let currentBg = { type:'solid', value:'#FFFFFF' };
  let currentData = { id: 'A001', title: '募驴丝晒', subtitle: 'UPF50+ 俑 女同' };
  let rows = [];
  let thumbs = [];
  let busy = false;
  function sampleRows(){
    return [
      { id:'A001', title:'募驴丝晒', subtitle:'UPF50+ 俑 女同' },
      { id:'A002', title:'旃揖', subtitle:'骞ぱ 痛' },
      { id:'A003', title:'票诨', subtitle:' 一 1.2L' },
      { id:'A004', title:'学', subtitle:'PPSU 摔 320ml' }
    ];
  }
  function parseCSV(text){
    const lines = text.split(/\r?\n/).filter(l => l.trim());
    if (!lines.length) return [];
    const headers = lines[0].split(',').map(s => s.trim());
    return lines.slice(1).map(line => {
      const cells = line.split(',').map(s => s.trim());
      const o = {}; headers.forEach((h, i) => o[h] = cells[i] || ''); return o;
    });
  }
  function isDarkBg(bg){
    let hex = bg.value;
    if (bg.type === 'gradient') hex = bg.colors[0];
    if (!hex || hex[0] !== '#') return false;
    const r = parseInt(hex.slice(1,3),16), g = parseInt(hex.slice(3,5),16), b = parseInt(hex.slice(5,7),16);
    return (r*0.299 + g*0.587 + b*0.114) < 128;
  }
  function wrapText(c, text, maxW){
    if (!text) return [];
    const out = []; let line = '';
    for (const ch of (text || '').split('')) {
      const t = line + ch;
      if (c.measureText(t).width > maxW && line) { out.push(line); line = ch; }
      else line = t;
    }
    if (line) out.push(line);
    return out;
  }
  function paintCanvas(bg, data){
    if (bg.type === 'solid') { ctx.fillStyle = bg.value; ctx.fillRect(0,0,W,H); }
    else {
      const g = ctx.createLinearGradient(0,0,W,H);
      g.addColorStop(0, bg.colors[0]); g.addColorStop(1, bg.colors[1]);
      ctx.fillStyle = g; ctx.fillRect(0,0,W,H);
    }
    const fontSize = parseInt(document.getElementById('biFontSize').value) || 72;
    const bold = document.getElementById('biBold').checked;
    const shadow = document.getElementById('biShadow').checked;
    const color = document.getElementById('biFontColor').value;
    ctx.fillStyle = color;
    ctx.textBaseline = 'top';
    if (shadow) { ctx.shadowColor = 'rgba(0,0,0,.4)'; ctx.shadowBlur = 14; ctx.shadowOffsetY = 4; }
    else ctx.shadowBlur = 0;
    ctx.font = (bold ? '800 ' : '400 ') + fontSize + 'px -apple-system, "PingFang SC", "Microsoft YaHei", sans-serif';
    const px = (parseInt(document.getElementById('biPosX').value) || 90) * (W/1080);
    const py = (parseInt(document.getElementById('biPosY').value) || 200) * (H/1920);
    let y = py;
    wrapText(ctx, data.title, W - px * 2).forEach(line => { ctx.fillText(line, px, y); y += fontSize * 1.25; });
    y += 30;
    ctx.font = '500 ' + Math.round(fontSize * 0.55) + 'px -apple-system, "PingFang SC", "Microsoft YaHei", sans-serif';
    ctx.fillStyle = isDarkBg(bg) ? 'rgba(255,255,255,.78)' : 'rgba(0,0,0,.6)';
    ctx.shadowBlur = 0;
    wrapText(ctx, data.subtitle, W - px * 2).forEach(line => { ctx.fillText(line, px, y); y += fontSize * 0.7; });
    ctx.font = '500 22px -apple-system, monospace';
    ctx.fillStyle = isDarkBg(bg) ? 'rgba(255,255,255,.55)' : 'rgba(0,0,0,.45)';
    ctx.fillText('#' + data.id, 40, H - 60);
  }
  function repaint(){ if (currentData) paintCanvas(currentBg, currentData); }
  document.querySelectorAll('.bi-swatch').forEach(s => {
    s.addEventListener('click', () => {
      document.querySelectorAll('.bi-swatch').forEach(x => x.classList.remove('active'));
      s.classList.add('active');
      const spec = s.dataset.bg;
      if (spec.startsWith('solid:')) currentBg = { type:'solid', value: spec.slice(6) };
      else if (spec.startsWith('gradient:')) currentBg = { type:'gradient', colors: spec.slice(9).split(',') };
      repaint();
    });
  });
  ['biFontSize','biBold','biShadow','biFontColor','biPosX','biPosY'].forEach(id => {
    const el = document.getElementById(id);
    if (el) el.addEventListener('input', repaint);
  });
  const csvBtn = document.getElementById('biCsvBtn');
  const csvIn = document.getElementById('biCsvInput');
  const loadSample = document.getElementById('biLoadSampleBtn');
  const status = document.getElementById('biCsvStatus');
  if (csvBtn && csvIn) csvBtn.addEventListener('click', () => csvIn.click());
  if (csvIn) csvIn.addEventListener('change', e => {
    const f = e.target.files[0]; if (!f) return;
    const reader = new FileReader();
    reader.onload = ev => loadRows(parseCSV(ev.target.result), f.name);
    reader.readAsText(f, 'utf-8');
  });
  if (loadSample) loadSample.addEventListener('click', () => loadRows(sampleRows(), 'sample.csv'));
  function loadRows(data, name){
    rows = data;
    status.textContent = '训 ' + name + ' (' + data.length + ' )';
    const headers = data.length ? Object.keys(data[0]) : ['id','title','subtitle'];
    ['biColId','biColTitle','biColSub'].forEach((id, idx) => {
      const sel = document.getElementById(id);
      const keys = ['id','title','subtitle'];
      sel.innerHTML = headers.map(h => '<option' + (h === keys[idx] ? ' selected' : '') + '>' + h + '</option>').join('');
    });
    thumbs = [];
    document.getElementById('biThumbList').innerHTML = '';
    if (data.length) { currentData = data[0]; repaint(); }
    document.getElementById('biPageInfo').textContent = data.length ? '1 / ' + data.length : '0 / 0';
  }
  const genBtn = document.getElementById('biGenerateBtn');
  const log = document.getElementById('biLog');
  function biLog(msg){
    const d = new Date();
    const t = d.toTimeString().slice(0,8);
    const line = document.createElement('div');
    line.innerHTML = '<span class="lt">' + t + '</span><i data-lucide="info" class="li" style="width:12px;height:12px"></i>' + msg;
    log.appendChild(line);
    log.scrollTop = log.scrollHeight;
    if (window.lucide) window.lucide.createIcons();
  }
  if (genBtn) genBtn.addEventListener('click', async () => {
    if (busy) return;
    if (!rows.length) { showToast('鹊 CSV 示'); return; }
    busy = true; genBtn.disabled = true;
    thumbs = []; document.getElementById('biThumbList').innerHTML = '';
    biLog('始 ' + rows.length + ' 图片');
    const fill = document.getElementById('biProgressFill');
    const text = document.getElementById('biProgressText');
    for (let i = 0; i < rows.length; i++){
      currentData = rows[i];
      paintCanvas(currentBg, currentData);
      const url = cv.toDataURL('image/png');
      thumbs.push(Object.assign({}, rows[i], { dataUrl: url }));
      const div = document.createElement('div'); div.className = 'bi-thumb' + (i === 0 ? ' active' : '');
      div.innerHTML = '<img src="' + url + '"><span class="bi-thumb-id">' + rows[i].id + '</span>';
      div.addEventListener('click', () => {
        document.querySelectorAll('.bi-thumb').forEach(x => x.classList.remove('active'));
        div.classList.add('active');
        currentData = rows[i]; repaint();
      });
      document.getElementById('biThumbList').appendChild(div);
      text.textContent = (i+1) + ' / ' + rows.length;
      fill.style.width = ((i+1) / rows.length * 100) + '%';
      document.getElementById('biPageInfo').textContent = (i+1) + ' / ' + rows.length;
      await new Promise(r => setTimeout(r, 30));
    }
    biLog('全');
    busy = false; genBtn.disabled = false;
    showToast(' ' + thumbs.length + ' 图片');
  });
  const zipBtn = document.getElementById('biZipBtn');
  if (zipBtn) zipBtn.addEventListener('click', async () => {
    if (!thumbs.length) { showToast('图片'); return; }
    biLog('诖 ZIP ...');
    const zip = new JSZip();
    for (const t of thumbs) {
      const b64 = t.dataUrl.split(',')[1];
      zip.file(t.id + '.png', b64, { base64: true });
    }
    const blob = await zip.generateAsync({ type:'blob' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'batch-images-' + Date.now() + '.zip';
    a.click();
    biLog('ZIP ');
    showToast('ZIP  (' + thumbs.length + ' )');
  });
  repaint();
})();

function showToast(msg){
  let t = document.getElementById('globalToast');
  if (!t) {
    t = document.createElement('div');
    t.id = 'globalToast';
    t.style.cssText = 'position:fixed;bottom:80px;left:50%;transform:translateX(-50%);background:#1F2024;color:#fff;padding:10px 18px;border-radius:999px;font-size:13px;z-index:100;box-shadow:0 6px 20px rgba(0,0,0,.18);opacity:0;transition:opacity .2s;pointer-events:none';
    document.body.appendChild(t);
  }
  t.textContent = msg;
  t.style.opacity = '1';
  clearTimeout(t._h);
  t._h = setTimeout(() => t.style.opacity = '0', 1800);
}

document.querySelectorAll('.tab').forEach(t => t.addEventListener('click', () => setTimeout(() => window.lucide && window.lucide.createIcons(), 50)));

if (window.lucide) window.lucide.createIcons();
});
